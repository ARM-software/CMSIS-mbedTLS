81 /* Slightly smaller way to check if tag is a string tag
82  * compared to canonical implementation. */
83 #define MBEDTLS_ASN1_IS_STRING_TAG(tag) \
84  ((unsigned int) (tag) < 32u && ( \
85  ((1u << (tag)) & ((1u << MBEDTLS_ASN1_BMP_STRING) | \
86  (1u << MBEDTLS_ASN1_UTF8_STRING) | \
